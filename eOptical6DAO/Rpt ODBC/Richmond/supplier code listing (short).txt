	Name: {@GetPrintStamp}
	Formula: "Printed on: " + Totext(CurrentDate,"MMM dd, yyyy") + "   " 
	+ Totext(CurrentTime,"hh:mm  tt") 
	+ "        Page : " + Totext(PageNumber,0)

	Name: {@GetCode}
	Formula: StringVar CodeVar;
	if isnull ({Supplier.SupplierCode}) then
	  CodeVar := ""
	else
	  CodeVar := Trim ({Supplier.SupplierCode}) + " : ";

	Name: {@GetTollFreePhone}
	Formula: WhilePrintingRecords;
	StringVar TollFreePhone1;
	StringVar TollFreePhone2;
	
	if (TollFreePhone1="" and TollFreePhone2="") then
	    ""
	  else
	  (if (TollFreePhone1="") then
	    TollFreePhone2
	   else
	   (if (TollFreePhone2="") then
	     TollFreePhone1
	   else
	    TollFreePhone1 + "  or  "  + TollFreePhone2));
	

	Name: {@BuildTollFreeData}
	Formula: WhilePrintingRecords;
	StringVar TollFreePhone1;
	StringVar TollFreePhone2;
	StringVar TollFreeFax1;
	StringVar TollFreeFax2;
	
	if isnull ({Supplier.TollFreePhone1}) then
	  TollFreePhone1:=""
	else 
	  (if ( "-" in {Supplier.TollFreePhone1}) then
	     TollFreePhone1:={Supplier.TollFreePhone1}
	   else
	     TollFreePhone1:=Left ({Supplier.TollFreePhone1},1) + "-" +
	                                 Mid ({Supplier.TollFreePhone1},2,3 ) + "-" +
	                                 Mid ({Supplier.TollFreePhone1}, 5,20 ));
	
	if isnull ({Supplier.TollFreePhone2}) then
	  TollFreePhone2:=""
	else 
	  (if ( "-" in {Supplier.TollFreePhone2}) then
	     TollFreePhone2:={Supplier.TollFreePhone2}
	   else
	     TollFreePhone2:=Left ({Supplier.TollFreePhone2},1) + "-" +
	                                 Mid ({Supplier.TollFreePhone2},2,3 ) + "-" +
	                                 Mid ({Supplier.TollFreePhone2}, 5,20 ));
	
	if isnull ({Supplier.TollFreeFax1}) then
	  TollFreeFax1:=""
	else 
	  (if ( "-" in {Supplier.TollFreeFax1}) then
	     TollFreeFax1:={Supplier.TollFreeFax1}
	   else
	     TollFreeFax1:=Left ({Supplier.TollFreeFax1},1) + "-" +
	                                 Mid ({Supplier.TollFreeFax1},2,3 ) + "-" +
	                                 Mid ({Supplier.TollFreeFax1}, 5,20 ));
	
	if isnull ({Supplier.TollFreeFax2}) then
	  TollFreeFax2:=""
	else 
	  (if ( "-" in {Supplier.TollFreeFax2}) then
	     TollFreeFax2:={Supplier.TollFreeFax2}
	   else
	     TollFreeFax2:=Left ({Supplier.TollFreeFax2},1) + "-" +
	                                 Mid ({Supplier.TollFreeFax2},2,3 ) + "-" +
	                                 Mid ({Supplier.TollFreeFax2}, 5,20 ));
