	Name: {@BuildSKU}
	Formula: WhilePrintingRecords;
	StringVar SKUs;
	StringVar TmpStr;
	
	if {TxHeader.Status} = "2"  then // Cancelled order
	  SKUs := "N/A"
	else
	(if not isnull ({TxDetails.SKU}) then 
	  (TmpStr:=" &" + chr(10)  + {TxDetails.SKU};
	   if (length(SKUs) + length(TmpStr)) <= 254 then
	     (if SKUs <> "" then 
	        SKUs := SKUs + TmpStr
	     else
	       SKUs := {TxDetails.SKU});
	   if not isnull ({Brand.BrandDescription}) then 
	   (TmpStr:= "/" + {Brand.BrandDescription};
	   if (length(SKUs) + length(TmpStr)) <= 254 then
	     SKUs := SKUs + TmpStr)));
	  

	Name: {@BuildPayMethod}
	Formula: WhilePrintingRecords;
	StringVar PayMethods;
	
	if {TxHeader.Status} = "2" then
	  PayMethods := "N/A"
	else if not isnull ({PaymentMethods.PaymentNameLong}) then 
	  if PayMethods <> "" and not ({PaymentMethods.PaymentNameLong} in PayMethods) then 
	    PayMethods := PayMethods + " &"  + chr(10) + {PaymentMethods.PaymentNameLong}
	  else
	  PayMethods :={PaymentMethods.PaymentNameLong};
	
	PayMethods;
