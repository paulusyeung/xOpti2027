SELECT
    Inventory.`Model`, Inventory.`Color`, Inventory.`Size`, Inventory.`SKU`,
    Brand.`BrandDescription`,
    RetailPrice.`RetailPrice`,
    Class.`ClassName`
FROM
    (((`Inventory` Inventory INNER JOIN `RetailPrice` RetailPrice ON
        Inventory.`SKU` = RetailPrice.`SKU`)
     INNER JOIN `Category` Category ON
        Inventory.`DepartmentCode` = Category.`DepartmentCode` AND
    Inventory.`ClassCode` = Category.`ClassCode` AND
    Inventory.`CategoryCode` = Category.`CategoryCode`)
     INNER JOIN `Brand` Brand ON
        Inventory.`BrandCode` = Brand.`BrandCode`)
     INNER JOIN `Class` Class ON
        Category.`DepartmentCode` = Class.`Department` AND
    Category.`ClassCode` = Class.`ClassCode`
