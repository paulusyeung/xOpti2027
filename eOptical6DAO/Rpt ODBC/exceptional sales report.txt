4.3 Other Formulas

	Name: {@GetPrintStamp}
	Formula: "Printed on: " + Totext(CurrentDate,"MMM dd, yyyy") + "   " 
	+ Totext(CurrentTime,"hh:mm  tt") 
	+ "        Page : " + Totext(PageNumber,0)

	Name: {@GetSubTitle}
	Formula: StringVar SubTitle := "Date between " +
	    Totext ({@AskDateFrom},"MMMM dd, yyyy") + " and "
	 + Totext ({@AskDateTo},"MMMM dd, yyyy")
	   + chr(10) +
	  "Gross Margin below " + totext({@AskPctBelow},0) + "% or above " +
	  totext({@AskPctAbove},0) + "%";

	Name: {@BuildSKU}
	Formula: WhilePrintingRecords;
	StringVar SKUs;
	
	if {TxHeader.Status} = "1" then
	  SKUs:= "N/A"
	else
	  (if not isnull ({TxDetails.SKU}) then 
	    // if SKUs <> "" then SKUs := SKUs + " &" + chr(10)  + {TxDetails.SKU}
	    // else
	       SKUs := {TxDetails.SKU};
	  if not isnull ({Brand.BrandDescription}) then 
	    SKUs := SKUs + "/" + {Brand.BrandDescription};)

	Name: {@GetInvoiceNumber}
	Formula: if {TxHeader.Status} = "1" then
	  {TxHeader.TxNumber} + chr(10) + "  * Refund *"
	else
	  {TxHeader.TxNumber};

	Name: {@GetSKU}
	Formula: WhilePrintingRecords;
	StringVar SKUs;

	Name: {@GetCustomerName}
	Formula: StringVar CustomerName := "";
	
	if not isnull({Customer.Salutation}) then CustomerName := trim({Customer.Salutation}) + " ";
	if isnull({Customer.FirstName}) then
	   CustomerName:= CustomerName + trim ({Customer.LastName})
	else
	  if isnull({Customer.LastName}) then
	    CustomerName :=CustomerName + trim ({Customer.FirstName})
	  else
	    CustomerName :=CustomerName + trim ({Customer.FirstName})+"  " + trim ({Customer.LastName});
	
	CustomerName;

	Name: {@GetCost}
	Formula: {TxDetails.Qty} * {TxDetails.UnitCost};

	Name: {@GetMargin}
	Formula: {TxDetails.Amount} - {@GetCost};

	Name: {@GetMarginPct}
	Formula: if {TxDetails.Amount}<>0 then
	   ({TxDetails.Amount} - {@GetCost} ) % {TxDetails.Amount}
	else
	  0;
