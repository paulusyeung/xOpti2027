	Name: {@BuildLeadingCode}
	Formula: WhilePrintingRecords;
	StringVar LeadingCode :="";
	
	if not isnull({Country.CountryCode}) then
	  LeadingCode:={Country.CountryPhoneCode} + "-";

	  (if Asc({City.CityPhoneCode}) in 48 to 57 then

	Name: {@GetAddress}
	Formula: StringVar Address := "";
	
	
	if Address <> "" then
	  (if Right(Address,1) in [";",","] then
	    Address := Address + " "
	  else
	    Address := Address + ", ");
	
	if not isnull ({City.Code}) then
	 (if not ({City.CityName} like "*/*") then
	  (if Address <> "" then
	     Address := Address + trim({City.CityName})
	  else
	     Address := trim({City.CityName})));
	
	if not isnull ({City.ProvinceCode}) then
	  (if not ({Province.ProvinceName} like "*/*") then
	    if Address <> "" then
	       Address := Address + ", " + trim ({Province.ProvinceName})
	    else
	       Address := trim({Province.ProvinceName}));
	
	if not isnull ({Supplier.PostCode}) then
	   if Address <> "" then 
	      Address :=  Address + ", " + trim({Supplier.PostCode})
	   else
	      Address := trim({Supplier.PostCode});
