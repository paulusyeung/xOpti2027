	Name: {@GetSales}
	Formula: StringVar OrderNo;
	
	if isnull({TxHeader.RefNumber}) then
	  OrderNo := "CashSale"
	else
	  OrderNo := {TxHeader.RefNumber};
	
	if {TxHeader.TxType} like "*C"  and OrderNo like "SA*" then   // Skip already exist in previous order
	  0
	else if {TxHeader.PaymentLink} =0 then  // This is a cash sale 
	  {TxHeader.TotalSale}
	else if {TxPayment.PaymentLineNo} <> 1 then   // This is a repeat due to slip payment
	  0
	else {TxHeader.TotalSale};

	Name: {@GetGST}
	Formula: dateVar cutoffDate := date(2010, 7, 1);
	
	if {@AskDateFrom} >= cutoffDate then
	  (if isnull({TxPayment.PaymentLineNo}) then
	    (if {TxHeader.TxType} <> "SA" then
	        {TxHeader.TaxA} + {TxHeader.TaxB}
	     else
	        0)
	  else if  {TxHeader.PaymentLink} =0 and {TxHeader.TxType} <> "SA" then
	   {TxHeader.TaxA} + {TxHeader.TaxB}
	  else if {TxPayment.PaymentLineNo}<>1 then
	    0
	  else if {TxHeader.TxType}<>"SA" then
	    {TxHeader.TaxA} + {TxHeader.TaxB}
	  else 0)
	else
	  (if isnull({TxPayment.PaymentLineNo}) then
	    (if {TxHeader.TxType} <> "SA" then
	        {TxHeader.TaxA}
	      else
	        0)
	  else if  {TxHeader.PaymentLink} =0 and {TxHeader.TxType} <> "SA" then
	   {TxHeader.TaxA}
	  else if {TxPayment.PaymentLineNo}<>1 then
	    0
	  else if {TxHeader.TxType}<>"SA" then
	    {TxHeader.TaxA}
	  else 0);

	Name: {@GetPST}
	Formula: dateVar cutoffDate := date(2010, 7, 1);
	
	if {@AskDateFrom} >= cutoffDate then
	  0
	else
	  (if isnull({TxPayment.PaymentLineNo}) then
	    (if {TxHeader.TxType} <> "SA"  then
	      {TxHeader.TaxB} 
	    else
	      0)
	  else if {TxHeader.PaymentLink}=0 and {TxHeader.TxType} <> "SA"  then
	   {TxHeader.TaxB} 
	  else if {TxPayment.PaymentLineNo} <>1 then
	    0
	  else if {TxHeader.TxType}<>"SA" then
	   {TxHeader.TaxB}
	  else 0);
	

	Name: {@GetGrossSales}
	Formula: if {TxHeader.TxType} like "*C*" and {TxHeader.PaymentLink} = 0 then
	  // This is a cash sale
	  {TxHeader.TotalAmount}
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "*C*" ) then
	  // This is a refund of sales
	  {TxHeader.TotalAmount}
	else if {TxHeader.TxType} like "*C" and {TxPayment.PaymentLineNo} < 2 then
	  //  This is invocie against confirmed order and 1st exist
	  {TxHeader.TotalAmount}
	else  0;

	Name: {@GetDepositToSales}
	Formula: if {TxHeader.TxType} like "*C" then
	  // This is a invoice and pervious deposit back to Sales
	  (if {TxPayment.PaymentLineNo} < 2 then  {TxHeader.PreviousPaid}
	    else  0);
	

	Name: {@GetBalanceReceived}
	Formula: if {TxHeader.PaymentLink} =0 then
	  // This is zero payment due to fully paid on order
	  0
	else if {TxHeader.TxType} like "*C" then
	  // This is a invoice
	     {TxPayment.TotalThisPay}
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "*C*") then
	  // This is a refund
	  {TxPayment.TotalThisPay}
	else
	  0;

	Name: {@GetDepositReceived}
	Formula: if {TxHeader.TxType} = "SA"  then 
	   // This is a deposit
	  (if {TxHeader.PaymentLink} = 0 then 0
	   else {TxPayment.TotalThisPay})
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "SA*") then
	  // This is a void order
	   {TxPayment.TotalThisPay}
	// else if {TxHeader.TxNumber} like "*C*" then
	  // This is a invoice 
	//  -{TxHeader.PreviousPaid}
	else
	  0;

	Name: {@GetRowTotal}
	Formula: {@GetDepositReceived}+{@GetBalanceReceived};  //+{@GetDepositToSales}
