4.3 Other Formulas

	Name: {@GetPrintStamp}
	Formula: "Printed on: " + Totext(CurrentDate,"MMM dd, yyyy") + "   " 
	+ Totext(CurrentTime,"hh:mm  tt") 
	+ "        Page : " + Totext(PageNumber,0)

	Name: {@GetSubTitle}
	Formula: StringVar SubTitle := "Date between " +
	    Totext ({@AskDateFrom},"MMMM dd, yyyy") + " and " + 
	    Totext ({@AskDateTo},"MMMM dd, yyyy");

	Name: {@GetInvoiceNumber}
	Formula: if {TxHeader.TxType} like "*C" then
	  "* " +{TxHeader.TxNumber}
	else if {TxHeader.RefNumber} like "*C*" then
	  "   " + {TxHeader.RefNumber}
	else
	  " ";     // others are N/A

	Name: {@GetOrderNumber}
	Formula: if {TxHeader.TxType} = "SA" then
	  "* "+ {TxHeader.TxNumber}
	else if {TxHeader.RefNumber} like "SA*" then
	 "   " + {TxHeader.RefNumber}
	else
	  " ";

	Name: {@GetParticulars}
	Formula: if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "*C*") then
	 "Refund Invoice as per " + {TxHeader.TxNumber}
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "SA*") then
	 "Void Order as per " + {TxHeader.TxNumber}
	else if {TxHeader.PaymentLink} > 1 then
	  "Spilt payment of " + totext({TxPayment.PaymentLineNo},0) +
	   "/" + totext({TxHeader.PaymentLink},0)
	else
	  " ";
	  

	Name: {@GetDepositToSales}
	Formula: if {TxHeader.TxType} like "*C" then
	  // This is a invoice and pervious deposit back to Sales
	  (if isnull({TxPayment.PaymentLineNo}) then
	      0
	  else if ({TxPayment.PaymentLineNo} = 1) then     // This is 1st Split
	      {TxHeader.PreviousPaid}
	  else
	      0);
	

	Name: {@GetBalanceReceived}
	Formula: if isnull({TxPayment.TotalThisPay}) then
	  0
	else if {TxHeader.PaymentLink} =0 then
	  // This is zero payment due to fully paid on order
	  0
	else if {TxHeader.TxType} like "*C" then
	  // This is a invoice
	     {TxPayment.TotalThisPay}
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "*C*") then
	  // This is a refund
	  {TxPayment.TotalThisPay}
	else
	  0;

	Name: {@GetDepositReceived}
	Formula: if isnull({TxPayment.TotalThisPay}) then
	  0
	else if {TxHeader.TxType} = "SA"  then 
	   // This is a deposit
	    {TxPayment.TotalThisPay}
	else if ({TxHeader.TxType} = "RS" and {TxHeader.RefNumber} like "SA*") then
	  // This is a void order
	     {TxPayment.TotalThisPay}
	//else if {TxHeader.TxNumber} like "*C*" then
	  // This is a invoice 
	//  -{TxHeader.PreviousPaid}
	else
	  0;

	Name: {@GetRowTotal}
	Formula: {@GetDepositReceived}+{@GetBalanceReceived};    // +{@GetDepositToSales}
